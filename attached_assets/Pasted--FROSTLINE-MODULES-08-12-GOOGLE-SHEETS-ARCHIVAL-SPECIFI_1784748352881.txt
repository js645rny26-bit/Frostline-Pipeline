## 08.4 IMPLEMENTATION LOGIC

```typescript
async function writeGoogleSheetsFeed(
  input: Module08Input
): Promise<Module08Output> {
  const results = {
    status: "success" as const,
    write_timestamp_utc: new Date().toISOString(),
    sheets_written: {} as any,
    errors: [] as ErrorLog[]
  };

  try {
    // 1. Write DAILY_MATCHUPS
    const dailyMatchupsRows = prepareDailyMatchupRows(input.normalized_slate);
    await clearRange(input.sheets_client, input.workbook_id, "DAILY_MATCHUPS!A3:AB32");
    const dmResult = await writeRows(
      input.sheets_client,
      input.workbook_id,
      "DAILY_MATCHUPS!A3:AB32",
      dailyMatchupsRows
    );
    results.sheets_written.daily_matchups = dmResult;

    // 2. Write TODAY_LINEUPS
    const todayLineupsRows = prepareTodayLineupsRows(input.normalized_slate);
    await clearRange(input.sheets_client, input.workbook_id, "TODAY_LINEUPS!A3:Q602");
    const tlResult = await writeRows(
      input.sheets_client,
      input.workbook_id,
      "TODAY_LINEUPS!A3:Q602",
      todayLineupsRows
    );
    results.sheets_written.today_lineups = tlResult;

    // 3. Write TEAM_FORM_INPUT (conditional on FanGraphs refresh)
    if (shouldRefreshTeamForm(input.normalized_slate)) {
      const teamFormRows = prepareTeamFormRows(input.normalized_slate);
      await clearRange(input.sheets_client, input.workbook_id, "TEAM_FORM_INPUT!A3:P62");
      const tfResult = await writeRows(
        input.sheets_client,
        input.workbook_id,
        "TEAM_FORM_INPUT!A3:P62",
        teamFormRows
      );
      results.sheets_written.team_form_input = tfResult;
    }

    // 4. Write BULLPEN_USAGE_DAILY
    const bullpenRows = prepareBullpenRows(input.normalized_slate);
    await clearRange(input.sheets_client, input.workbook_id, "BULLPEN_USAGE_DAILY!A2:U31");
    // Also regenerate date headers
    await writeRows(
      input.sheets_client,
      input.workbook_id,
      "BULLPEN_USAGE_DAILY!D1:I1",
      [generateBullpenDateHeaders(input.normalized_slate)]
    );
    const buResult = await writeRows(
      input.sheets_client,
      input.workbook_id,
      "BULLPEN_USAGE_DAILY!A2:U31",
      bullpenRows
    );
    results.sheets_written.bullpen_usage_daily = buResult;

    // 5. Write RUN_ENVIRONMENT (preserve operator notes in column R)
    const runEnvRows = prepareRunEnvironmentRows(input.normalized_slate);
    await clearRange(input.sheets_client, input.workbook_id, "RUN_ENVIRONMENT!A2:Q31");
    const reResult = await writeRows(
      input.sheets_client,
      input.workbook_id,
      "RUN_ENVIRONMENT!A2:Q31",
      runEnvRows
    );
    results.sheets_written.run_environment = reResult;

    return results;

  } catch (error) {
    results.status = "failure";
    results.errors.push({
      module: "08_google_sheets_feed",
      error: error.message,
      timestamp: new Date().toISOString()
    });
    return results;
  }
}

// Helper: prepare row data from normalized slate
function prepareDailyMatchupRows(slate: NormalizedSlate): any[][] {
  return slate.games.map(game => [
    game.date,
    game.legacy_game_id,
    game.away_team.team_abbr,
    game.home_team.team_abbr,

    game.game_status.codedGameState,

    "valid"
  ]);
```

## 08.5 ERROR HANDLING & FALLBACKS

| Scenario | Action |
|---|---|
| **Single sheet write fails** | Log ERROR for that sheet; continue with others (PARTIAL_FAILURE status) |
| **Protected column overwrite attempted** | Catch by Google Sheets API; log ERROR; do not write that row |
| **Row count mismatch** | Truncate or pad to expected row count; log WARNING |
| **Timeout on write** | Retry 3 times; if still fails, mark sheet as FAILURE |
| **All writes fail** | Return status="failure"; do not proceed to Module 09 |

## 09.3 VERIFICATION LOGIC

```typescript
async function verifyRecalculation(
  input: Module09Input
): Promise<Module09Output> {
  const startTime = Date.now();
  const output: Module09Output = {
    status: "verified",
    verification_timestamp_utc: new Date().toISOString(),
    checks: {
      game_integration: { status: "verified", expected_rows: 0, actual_rows: 0, formula_errors: [] },
      game_summary: { status: "verified", expected_rows: 0, actual_rows: 0, formula_errors: [] },
      consistency_check: { status: "consistent", read_1_timestamp: "", read_2_timestamp: "", diff_seconds: 0 }
    },
    recalculation_time_ms: 0
  };

  // Expected: 2 rows per game (away + home)
  const expectedIntegrationRows = input.expected_game_count * 2;
  const expectedSummaryRows = input.expected_game_count;

  for (let attempt = 0; attempt < input.max_retries; attempt++) {
    try {
      // Read GAME_INTEGRATION column A (game IDs)
      const integrationRange = await input.sheets_client.getRange(
        input.workbook_id,
        "GAME_INTEGRATION!A:A"
      );
      const integrationRows = integrationRange.values?.length || 0;

      // Check for formula errors in key columns
      const integrationErrors = checkForFormulaErrors(integrationRange);

      // Read GAME_SUMMARY column A
      const summaryRange = await input.sheets_client.getRange(
        input.workbook_id,
        "GAME_SUMMARY!A:A"
      );
      const summaryRows = summaryRange.values?.length || 0;
      const summaryErrors = checkForFormulaErrors(summaryRange);

      output.checks.game_integration = {
        status: integrationErrors.length > 0 ? "error" : "verified",
        expected_rows: expectedIntegrationRows,
        actual_rows: integrationRows,
        formula_errors: integrationErrors
      };

      output.checks.game_summary = {
        status: summaryErrors.length > 0 ? "error" : "verified",
        expected_rows: expectedSummaryRows,
        actual_rows: summaryRows,
        formula_errors: summaryErrors
      };

      // If row counts match and no errors, proceed to consistency check
      if (
        integrationRows >= expectedIntegrationRows &&
        summaryRows >= expectedSummaryRows &&
        integrationErrors.length === 0 &&
        summaryErrors.length === 0
      ) {
        // Consistency check: read twice, verify no change
        const read1 = await input.sheets_client.getRange(
          input.workbook_id,
          "GAME_SUMMARY!B:C"  // Sample columns
        );
        const read1Time = new Date().toISOString();

        await sleep(1000);

        const read2 = await input.sheets_client.getRange(
          input.workbook_id,
          "GAME_SUMMARY!B:C"
        );
        const read2Time = new Date().toISOString();

        const isConsistent = JSON.stringify(read1) === JSON.stringify(read2);

        output.checks.consistency_check = {
          status: isConsistent ? "consistent" : "inconsistent",
          read_1_timestamp: read1Time,
          read_2_timestamp: read2Time,
          diff_seconds: 1
        };

        if (isConsistent) {
          output.status = "verified";
          output.recalculation_time_ms = Date.now() - startTime;
          return output;
        }
      }

      // If row counts don't match, retry
      if (
        integrationRows < expectedIntegrationRows ||
        summaryRows < expectedSummaryRows
      ) {
        if (attempt < input.max_retries - 1) {
          await sleep(input.retry_delay_ms);
          continue;
        } else {
          output.status = "incomplete";
          output.recalculation_time_ms = Date.now() - startTime;
          return output;
        }
      }

    } catch (error) {
      if (attempt < input.max_retries - 1) {
        await sleep(input.retry_delay_ms);
        continue;
      } else {
        output.status = "error";
        output.recalculation_time_ms = Date.now() - startTime;
        return output;
      }
    }
  }

  output.status = "timeout";
  output.recalculation_time_ms = Date.now() - startTime;
  return output;
}

function checkForFormulaErrors(range: any): string[] {
  const errors: string[] = [];
  if (!range.values) return errors;

  range.values.forEach((row: any[], idx: number) => {
    row.forEach((cell: any, col: number) => {
      if (typeof cell === "string") {
        if (
          cell.startsWith("#REF!") ||
          cell.startsWith("#VALUE!") ||
          cell.startsWith("#NAME?") ||
          cell.startsWith("#DIV/0!") ||
          cell.startsWith("#N/A")
        ) {
          errors.push(`${String.fromCharCode(65 + col)}${idx + 1}: ${cell}`);
        }
      }
    });
  });

  return errors;
}

function sleep(ms: number): Promise<void> {
  return new Promise(resolve => setTimeout(resolve, ms));
}
```

## 10.3 SEEDING LOGIC (TWO-STAGE)

```typescript
async function seedSlateInput(
  input: Module10Input
): Promise<Module10Output> {
  const output: Module10Output = {
    status: "success",
    seeding_timestamp_utc: new Date().toISOString(),
    games_seeded: { new_games: 0, updated_games: 0, total_games: 0 },
    rows_written: 0,
    seed_results: [],
    errors: []
  };

  try {
    // Index existing SLATE_INPUT rows by Game_ID
    const existingByGameId = new Map<string, any>();
    for (const row of input.current_slate_input_rows) {
      const gameId = row["Game_ID"] || row[0];  // Column A
      if (gameId) {
        existingByGameId.set(gameId, row);
      }
    }

    const seededRows: any[] = [];

    for (const game of input.normalized_slate.games) {
      const gameId = game.legacy_game_id;
      let seededRow: any;

      if (existingByGameId.has(gameId)) {
        // EXISTING GAME: Update model fields, preserve operator fields
        const existingRow = existingByGameId.get(gameId);
        seededRow = { ...existingRow };  // Start with existing row

        // MODEL FIELDS: Extract from GAME_INTEGRATION and GAME_SUMMARY
        const modelData = extractModelData(
          gameId,
          input.game_integration_data,
          input.game_summary_data
        );

        // Update model fields (example fields)
        seededRow["Truth_Family"] = modelData.truth_family;
        seededRow["Event_Score"] = modelData.event_score;
        seededRow["Run_Conversion"] = modelData.run_conversion;
        seededRow["Early_Conversion"] = modelData.early_conversion;
        seededRow["Contact_Quality"] = modelData.contact_quality;
        seededRow["Truth_Score"] = modelData.truth_score;
        seededRow["Vehicle_Score"] = modelData.vehicle_score;
        seededRow["Confirmation_Gate"] = modelData.confirmation_gate;
        seededRow["Execution_Status"] = modelData.execution_status;

        // OPERATOR FIELDS: DO NOT OVERWRITE
        // seededRow["Candidate_Vehicle"] stays as-is
        // seededRow["Line"] stays as-is
        // seededRow["Odds"] stays as-is
        // seededRow["Notes"] stays as-is
        // seededRow["Owner"] stays as-is
        // etc.

        output.games_seeded.updated_games++;

        output.seed_results.push({
          legacy_game_id: gameId,
          action: "updated",
          model_fields_refreshed: [
            "Truth_Family",
            "Event_Score",
            "Run_Conversion",
            "Truth_Score",
            "Vehicle_Score"
          ],
          operator_fields_preserved: [
            "Candidate_Vehicle",
            "Line",
            "Odds",
            "Notes",
            "Owner"
          ]
        });

      } else {
        // NEW GAME: Seed with model data + operator defaults
        seededRow = {
          "Game_ID": gameId,
          "Date": game.date,
          "Matchup": `${game.away_team.team_abbr} @ ${game.home_team.team_abbr}`,
          "Target": "Game/Side",
          "Opposing_Starter": game.home_pitcher.name,
          // Model fields
          ...extractModelData(gameId, input.game_integration_data, input.game_summary_data),
          // Operator defaults
          "Candidate_Vehicle": "TBD",
          "Line": null,
          "Odds": null,
          "Market_Available": false,
          "Kill_Flag": false,
          "Notes": "Seeded by pipeline; awaiting vehicle selection and market data",
          "Owner": "Pending",
          "Manual_Kill_Override": false,
          "Model_Freeze_Reason": null
        };

        output.games_seeded.new_games++;

        output.seed_results.push({
          legacy_game_id: gameId,
          action: "created",
          model_fields_refreshed: Object.keys(extractModelData(gameId, input.game_integration_data, input.game_summary_data)),
          operator_fields_preserved: [
            "Candidate_Vehicle (default: TBD)",
            "Line (default: null)",
            "Owner (default: Pending)"
          ]
        });
      }

      seededRows.push(seededRow);
    }

    // Write seeded rows back to SLATE_INPUT
    // Match by Game_ID for existing, append for new
    const writeResult = await writeSeedSlateInput(
      input.sheets_client,
      input.workbook_id,
      seededRows
    );

    output.rows_written = seededRows.length;
    output.games_seeded.total_games = seededRows.length;

    if (!writeResult.success) {
      output.status = "failure";
      output.errors.push({
        module: "10_slate_input_seeding",
        error: writeResult.error,
        timestamp: new Date().toISOString()
      });
    }

    return output;

  } catch (error) {
    output.status = "failure";
    output.errors.push({
      module: "10_slate_input_seeding",
      error: error.message,
      timestamp: new Date().toISOString()
    });
    return output;
  }
}
```

```typescript
async function extractOutputBoards(
  input: Module11Input
): Promise<Module11Output> {
  const output: Module11Output = {
    status: "success",
    extraction_timestamp_utc: new Date().toISOString(),
    slate_board: [],
    active_board_snapshot: [],
    core_count: 0,
    not_core_count: 0
  };

  try {
    // Read SLATE_BOARD (entire sheet or specific range)
    const slateBoard = await input.sheets_client.getRange(
      input.workbook_id,
      "SLATE_BOARD!A:Z"
    );

    // Parse SLATE_BOARD rows (skip header)
    if (slateBoard.values) {
      for (let i = 1; i < slateBoard.values.length; i++) {
        const row = slateBoard.values[i];
        
        const finalDecision = row[columnIndex("Final_Decision")] as "CORE" | "NOT_CORE";
        
        output.slate_board.push({
          gamePk: row[columnIndex("GamePk")],
          legacy_game_id: row[columnIndex("Game_ID")],
          matchup: row[columnIndex("Matchup")],
          final_decision: finalDecision,
          truth_score: row[columnIndex("Truth_Score")] || 0,
          vehicle_score: row[columnIndex("Vehicle_Score")] || 0,
          best_vehicle_decision: row[columnIndex("Best_Vehicle_Decision")],
          not_core_reason: finalDecision === "NOT_CORE" ? row[columnIndex("NOT_CORE_Reason")] : undefined,
          confidence: determineConfidence(row[columnIndex("Truth_Score")])
        });

        if (finalDecision === "CORE") output.core_count++;
        if (finalDecision === "NOT_CORE") output.not_core_count++;
      }
    }

    // Read ACTIVE_BOARD_SNAPSHOT
    const activeBoard = await input.sheets_client.getRange(
      input.workbook_id,
      "ACTIVE_BOARD_SNAPSHOT!A:Z"
    );

    if (activeBoard.values) {
      for (let i = 1; i < activeBoard.values.length; i++) {
        const row = activeBoard.values[i];

        output.active_board_snapshot.push({
          date: row[columnIndex("Date")],
          game_id: row[columnIndex("Game_ID")],
          matchup: row[columnIndex("Matchup")],
          decision: row[columnIndex("Decision")] as "CORE" | "NOT_CORE",
          side_lean: row[columnIndex("Side_Lean")],
          total_lean: row[columnIndex("Total_Lean")],
          away_confidence: row[columnIndex("Away_Confidence")],
          home_confidence: row[columnIndex("Home_Confidence")]
        });
      }
    }

    return output;

  } catch (error) {
    output.status = "failure";
    return output;
  }
}

function columnIndex(columnName: string): number {
  // Map column names to indices
  const columns: Record<string, number> = {
    "GamePk": 0,
    "Game_ID": 1,
    "Matchup": 2,
    "Final_Decision": 10,
    "Truth_Score": 15,
    "Vehicle_Score": 16,
    "Best_Vehicle_Decision": 17,
    "NOT_CORE_Reason": 18,
    // ... add all columns from your SLATE_BOARD
  };
  
  return columns[columnName] || -1;
}

function determineConfidence(truthScore: number): "high" | "moderate" | "low" {
  if (truthScore >= 80) return "high";
  if (truthScore >= 65) return "moderate";
  return "low";
}
```
