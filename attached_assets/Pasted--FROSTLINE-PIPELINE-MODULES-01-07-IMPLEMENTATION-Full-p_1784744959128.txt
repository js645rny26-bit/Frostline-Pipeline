```python
"""
Shared utilities for Frostline pipeline modules.
"""

import json
import os
from datetime import datetime, timedelta
from pathlib import Path

def load_json(filepath):
    """Load JSON file from local filesystem."""
    try:
        with open(filepath, 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        return None
    except json.JSONDecodeError as e:
        print(f"JSON decode error in {filepath}: {e}")
        return None

def save_json(filepath, data):
    """Save JSON file to local filesystem."""
    os.makedirs(os.path.dirname(filepath), exist_ok=True)
    with open(filepath, 'w') as f:
        json.dump(data, f, indent=2)

def get_config(key):
    """Load configuration file."""
    config_path = f"config/{key}.json"
    return load_json(config_path) or {}

def get_validation_rules():
    """Load validation rules."""
    return get_config("validation_rules")

def get_source_mappings():
    """Load source mappings (teams, IDs)."""
    return get_config("source_mappings")

def get_stadium_coords():
    """Load stadium coordinates."""
    return get_config("stadium_coordinates")

def celsius_to_fahrenheit(celsius):
    """Convert Celsius to Fahrenheit."""
    if celsius is None:
        return None
    return round((celsius * 9/5) + 32, 1)

def kmh_to_mph(kmh):
    """Convert km/h to mph."""
    if kmh is None:
        return None
    return round(kmh * 0.621371, 1)

def date_diff(date_str, from_date=None):
    """Days between two dates."""
    from_date = from_date or datetime.now().strftime("%Y-%m-%d")
    try:
        d1 = datetime.strptime(date_str, "%Y-%m-%d")
        d2 = datetime.strptime(from_date, "%Y-%m-%d")
        return abs((d2 - d1).days)
    except:
        return 999

def log_message(level, message, context=""):
    """Log message with timestamp."""
    timestamp = datetime.utcnow().isoformat() + "Z"
    log_entry = f"[{timestamp}] {level:8} {context:20} {message}"
    print(log_entry)
    return log_entry

def normalize_team_abbr(abbr, mappings=None):
    """Normalize team abbreviation using mappings."""
    if not mappings:
        mappings = get_source_mappings()
    
    # Try to find by canonical abbreviation
    for team_id, team_info in mappings.get("teams", {}).items():
        if team_info.get("canonical_abbr") == abbr.upper():
            return abbr.upper()
        if abbr.upper() in team_info.get("aliases", []):
            return team_info.get("canonical_abbr")
    
    return abbr.upper()
```

"""
Module 01: MLB StatsAPI
Fetch authoritative schedule, game identity, probable pitchers, lineups, game status.
"""

import requests
from datetime import datetime
from utils import log_message, get_source_mappings, normalize_team_abbr

STATSAPI_BASE = "https://statsapi.mlb.com/api/v1"
TIMEOUT_SECONDS = 10

def fetch_mlb_schedule(date_str, sport_id=1):
    """
    Fetch complete MLB schedule for a given date.
    
    Args:
        date_str: ISO date (YYYY-MM-DD)
        sport_id: 1 for MLB
    
    Returns:
        dict with schedule + probable pitchers
    """
    
    log_message("INFO", f"Fetching MLB schedule for {date_str}", "MODULE_01")
    
    try:
        url = f"{STATSAPI_BASE}/schedule"
        params = {
            "sportId": sport_id,
            "date": date_str,
            "hydrate": "probablePitcher(note)"
        }
        
        response = requests.get(url, params=params, timeout=TIMEOUT_SECONDS)
        response.raise_for_status()
        data = response.json()
        
        games = []
        for date_entry in data.get("dates", []):
            games.extend(date_entry.get("games", []))
        
        log_message("INFO", f"Retrieved {len(games)} games from StatsAPI", "MODULE_01")
        
        # Normalize each game
        normalized_games = []
        for game in games:
            normalized_game = normalize_game_record(game)
            normalized_games.append(normalized_game)
        
        return {
            "retrieval_timestamp_utc": datetime.utcnow().isoformat() + "Z",
            "date": date_str,
            "total_games": len(normalized_games),
            "games": normalized_games,
            "status": "success"
        }
    
    except requests.Timeout:
        log_message("ERROR", "MLB StatsAPI timeout (>10s)", "MODULE_01")
        return {
            "error": "StatsAPI timeout",
            "status": "failed",
            "games": []
        }
    
    except requests.RequestException as e:
        log_message("ERROR", f"MLB StatsAPI request failed: {e}", "MODULE_01")
        return {
            "error": str(e),
            "status": "failed",
            "games": []
        }
    
    except Exception as e:
        log_message("ERROR", f"Unexpected error: {e}", "MODULE_01")
        return {
            "error": str(e),
            "status": "failed",
            "games": []
        }

def normalize_game_record(raw_game):
    """Normalize a single game record from StatsAPI."""
    
    mappings = get_source_mappings()
    
    away_team = raw_game.get("teams", {}).get("away", {})
    home_team = raw_game.get("teams", {}).get("home", {})
    
    away_team_id = away_team.get("team", {}).get("id")
    home_team_id = home_team.get("team", {}).get("id")
    
    # Get canonical abbreviations
    away_team_info = mappings.get("teams", {}).get(str(away_team_id), {})
    home_team_info = mappings.get("teams", {}).get(str(home_team_id), {})
    
    away_abbr = away_team_info.get("canonical_abbr", "UNK")
    home_abbr = home_team_info.get("canonical_abbr", "UNK")
    
    # Build legacy game ID
    game_date = raw_game.get("gameDateTime", "").split("T")[0].replace("-", "")
    legacy_game_id = f"{game_date}_{away_abbr}_{home_abbr}"
    
    # Extract probable pitchers
    away_prob = away_team.get("probablePitcher", {})
    home_prob = home_team.get("probablePitcher", {})
    
    return {
        "gamePk": raw_game.get("gamePk"),
        "legacy_game_id": legacy_game_id,
        "gameDateTime": raw_game.get("gameDateTime"),
        "gameLocalTime": raw_game.get("officialDate") + "T" + raw_game.get("gameDateTime", "").split("T")[1] if "T" in raw_game.get("gameDateTime", "") else None,
        "venue": {
            "id": raw_game.get("venue", {}).get("id"),
            "name": raw_game.get("venue", {}).get("name"),
            "timeZone": raw_game.get("venue", {}).get("timeZone", "America/New_York")
        },
        "teams": {
            "away": {
                "team": {
                    "id": away_team_id,
                    "name": away_team.get("team", {}).get("name"),
                    "abbreviation": away_abbr
                },
                "probablePitcher": {
                    "id": away_prob.get("id"),
                    "fullName": away_prob.get("fullName"),
                    "hand": away_prob.get("handedness", {}).get("code") if isinstance(away_prob.get("handedness", {}), dict) else None
                }
            },
            "home": {
                "team": {
                    "id": home_team_id,
                    "name": home_team.get("team", {}).get("name"),
                    "abbreviation": home_abbr
                },
                "probablePitcher": {
                    "id": home_prob.get("id"),
                    "fullName": home_prob.get("fullName"),
                    "hand": home_prob.get("handedness", {}).get("code") if isinstance(home_prob.get("handedness", {}), dict) else None
                }
            }
        },
        "status": {
            "abstractGameState": raw_game.get("status", {}).get("abstractGameState"),
            "detailedState": raw_game.get("status", {}).get("detailedState"),
            "codedGameState": raw_game.get("status", {}).get("codedGameState")
        },
        "doubleheaderStatus": raw_game.get("doubleheaderStatus", "NONE"),
        "gameNumber": raw_game.get("gameNumber"),
        "seriesGameNumber": raw_game.get("seriesGameNumber")
    }

"""
Module 02: Statcast Workload
Fetch recent pitcher usage, metrics, and workload data.
"""

import requests
from datetime import datetime, timedelta
from utils import log_message

BASEBALL_SAVANT_CSV_ENDPOINT = "https://baseballsavant.mlb.com/statcast_search/csv"
TIMEOUT_SECONDS = 15

def fetch_pitcher_workload(pitcher_ids, end_date, lookback_days=30):
    """
    Fetch recent pitcher workload from Statcast.
    
    Args:
        pitcher_ids: List of MLB pitcher IDs
        end_date: Date through which to fetch (YYYY-MM-DD)
        lookback_days: How far back to look (default 30)
    
    Returns:
        dict with pitcher workload data
    """
    
    log_message("INFO", f"Fetching workload for {len(pitcher_ids)} pitchers", "MODULE_02")
    
    if not pitcher_ids:
        log_message("WARNING", "No pitcher IDs provided", "MODULE_02")
        return {"pitchers": [], "status": "no_pitchers"}
    
    pitchers_data = {}
    
    for pitcher_id in pitcher_ids:
        if not pitcher_id:
            continue
        
        try:
            workload = fetch_single_pitcher_workload(pitcher_id, end_date, lookback_days)
            if workload:
                pitchers_data[pitcher_id] = workload
        
        except Exception as e:
            log_message("WARNING", f"Failed to fetch pitcher {pitcher_id}: {e}", "MODULE_02")
            pitchers_data[pitcher_id] = {
                "playerId": pitcher_id,
                "status": "fetch_error",
                "error": str(e)
            }
    
    log_message("INFO", f"Retrieved workload for {len(pitchers_data)} pitchers", "MODULE_02")
    
    return {
        "retrieval_timestamp_utc": datetime.utcnow().isoformat() + "Z",
        "retrieval_source": "statcast",
        "data_through_date": (datetime.strptime(end_date, "%Y-%m-%d") - timedelta(days=1)).strftime("%Y-%m-%d"),
        "pitchers": list(pitchers_data.values()),
        "status": "success" if pitchers_data else "no_data"
    }

def fetch_single_pitcher_workload(pitcher_id, end_date, lookback_days):

            "pitcher_id": pitcher_id,

        }

            BASEBALL_SAVANT_CSV_ENDPOINT,
            params=params,
            timeout=TIMEOUT_SECONDS
        )
        response.raise_for_status()
        
        # Parse CSV response
        lines = response.text.strip().split('\n')
        if len(lines) < 2:
            # No data
            return {
                "playerId": pitcher_id,
                "status": "no_games_in_window",
                "rolling_stats": {}
            }
        
        # Extract pitcher name from first data row (if available)
        header = lines[0].split(',')
        pitcher_name_idx = header.index("pitcher") if "pitcher" in header else -1
        pitcher_name = "Unknown"
        
        if pitcher_name_idx >= 0 and len(lines) > 1:
            first_row = lines[1].split(',')

                pitcher_name = first_row[pitcher_name_idx]

            "playerId": pitcher_id,
            "name": pitcher_name,
            "status": "active",
            "rolling_stats": rolling_stats,
            "recent_games_count": len(lines) - 1
        }
    
    except requests.Timeout:
        return {

        game_pk = game.get("gamePk")

        away_pitcher_classification = classify_single_pitcher(
            away_pitcher_id,
            away_pitcher_name,
            workload_by_id.get(away_pitcher_id),
            away_hand
        )
        
        # Classify home pitcher
        home_pitcher_classification = classify_single_pitcher(
            home_pitcher_id,
            home_pitcher_name,
            workload_by_id.get(home_pitcher_id),
            home_hand
        )
        
        if away_pitcher_classification["role"] == "UNRESOLVED":
            unresolved_count += 1
        if home_pitcher_classification["role"] == "UNRESOLVED":
            unresolved_count += 1
        
        classified_games.append({

            "legacy_game_id": game.get("legacy_game_id"),

            "home_pitcher": home_pitcher_classification
        })

"""

Fetch weather forecasts for each stadium.
"""

import requests
from datetime import datetime

OPEN_METEO_API = "https://api.open-meteo.com/v1/forecast"
TIMEOUT_SECONDS = 8

def fetch_weather_forecasts(manifest):
    """
    Fetch weather forecasts for all games.
    
    Args:
        manifest: Normalized schedule from Module 01
    
    Returns:
        dict with weather for each game
    """
    
    log_message("INFO", f"Fetching weather for {manifest.get('total_games', 0)} games", "MODULE_04")
    
    stadium_coords = get_stadium_coords()
    
    weather_data = []
    success_count = 0
    fallback_count = 0
    
    for game in manifest.get("games", []):
        game_pk = game.get("gamePk")
        venue_name = game.get("venue", {}).get("name")
        game_time = game.get("gameDateTime")
        
        # Look up stadium coordinates
        coords = stadium_coords.get(venue_name)
        
        if not coords:
            log_message("WARNING", f"Stadium coords not found for {venue_name}", "MODULE_04")
            weather_data.append({
                "gamePk": game_pk,
                "venue": venue_name,
                "status": "missing_coords",
                "weather": None
            })
            fallback_count += 1
            continue
        
        try:
            weather = fetch_open_meteo_forecast(coords, game_time)
            if weather:
                weather_data.append({
                    "gamePk": game_pk,
                    "venue": venue_name,
                    "status": "success",
                    "weather": weather
                })
                success_count += 1
            else:
                # Fallback to defaults
                weather_data.append({
                    "gamePk": game_pk,
                    "venue": venue_name,
                    "status": "fallback",
                    "weather": get_default_weather()
                })
                fallback_count += 1
        
        except Exception as e:
            log_message("WARNING", f"Weather fetch failed for {venue_name}: {e}", "MODULE_04")
            weather_data.append({
                "gamePk": game_pk,
                "venue": venue_name,
                "status": "fallback",
                "weather": get_default_weather()
            })
            fallback_count += 1
    
    log_message("INFO", f"Weather: {success_count} success, {fallback_count} fallback", "MODULE_04")
    
    return {
        "retrieval_timestamp_utc": datetime.utcnow().isoformat() + "Z",
        "retrieval_source": "open_meteo_with_fallback",
        "games": weather_data,
        "status": "success"
    }

def fetch_open_meteo_forecast(coords, game_time):
    """Fetch single game weather from Open-Meteo."""
    
    try:
        params = {
            "latitude": coords["latitude"],
            "longitude": coords["longitude"],
            "hourly": "temperature_2m,relative_humidity_2m,precipitation_probability,wind_speed_10m,wind_direction_10m",
            "timezone": coords.get("timezone", "UTC"),
            "forecast_days": 7
        }
        
        response = requests.get(OPEN_METEO_API, params=params, timeout=TIMEOUT_SECONDS)
        response.raise_for_status()
        data = response.json()
        
        # Find hourly entry closest to game time
        hourly_data = data.get("hourly", {})
        times = hourly_data.get("time", [])
        
        if not times:
            return None
        
        # Simple: use first entry (in production, would match to game time hour)
        idx = 0
        
        return {
            "temperature_f": celsius_to_fahrenheit(hourly_data.get("temperature_2m", [None])[idx]),
            "humidity_pct": hourly_data.get("relative_humidity_2m", [None])[idx],
            "wind_speed_mph": kmh_to_mph(hourly_data.get("wind_speed_10m", [None])[idx]),
            "wind_direction_degrees": hourly_data.get("wind_direction_10m", [None])[idx],
            "precipitation_probability_pct": hourly_data.get("precipitation_probability", [None])[idx],
            "data_quality": "good"
        }
    
    except Exception as e:
        log_message("WARNING", f"Open-Meteo API error: {e}", "MODULE_04")
        return None

            log_message("INFO", f"Using cached team splits (age: {cache_age} days)", "MODULE_05")

                **cached_data,

                "freshness_status": "valid"

        return {
            **fresh_data,
            "retrieval_source": "fangraphs_live",

    min_games = rules.get("game_count", {}).get("min_expected", 13)

        if game.get("home_pitcher", {}).get("role") == "UNRESOLVED":

    for game in games:
        if game.get("doubleheader_status") != "NONE":
            info_notes.append(f"Doubleheader game: {game.get('legacy_game_id')}")
    
    status = "PASS" if not critical_failures else "FAIL"

        fallback_count = sum(1 for g in weather.get("games", []) if g.get("status") == "fallback")
        
        print(f"✓ Retrieved weather: {success_count} live, {fallback_count} fallback")
        test_results["modules"]["04_open_meteo"] = {
            "status": "PASS",
            "live": success_count,
            "fallback": fallback_count
        }
    
    except Exception as e:
        print(f"✗ FAILED: {e}")
        test_results["modules"]["04_open_meteo"] = {
            "status": "FAIL",
            "error": str(e)
        }
        return False
    
    # STEP 5: Fetch team splits
    print("\n[5/7] FanGraphs Team Splits")
    print("-" * 70)
    try:
        splits = fetch_team_splits_with_fallback()
        
        if splits.get("status") == "failed":
            raise Exception(splits.get("error", "Unknown error"))
        
        team_count = len(splits.get("teams", []))
        source = splits.get("retrieval_source", "unknown")
        freshness = splits.get("freshness_status", "unknown")
        
        print(f"✓ Retrieved splits for {team_count // 2} teams")
        print(f"  - Source: {source}")
        print(f"  - Freshness: {freshness}")
        test_results["modules"]["05_fangraphs"] = {
            "status": "PASS",
            "teams": team_count // 2,
            "source": source
        }
    
    except Exception as e:
        print(f"✗ FAILED: {e}")
        test_results["modules"]["05_fangraphs"] = {
            "status": "FAIL",
            "error": str(e)
        }
        return False
    
    # STEP 6: Normalize slate
    print("\n[6/7] Normalization")
    print("-" * 70)
    try:
        normalized = normalize_slate(manifest, workload, roles, weather, splits)
        
        if normalized.get("status") != "success":
            raise Exception(normalized.get("error", "Unknown error"))
        
        game_count = len(normalized.get("games", []))
        
        print(f"✓ Normalized {game_count} games")
        test_results["modules"]["06_normalization"] = {
            "status": "PASS",
            "games_normalized": game_count
        }
    
    except Exception as e:
        print(f"✗ FAILED: {e}")
        test_results["modules"]["06_normalization"] = {
            "status": "FAIL",
            "error": str(e)
        }
        return False
    
    # STEP 7: Validate
    print("\n[7/7] Validation")
    print("-" * 70)
    try:
        validation = validate_normalized_slate(normalized)
        
        status = validation.get("status", "UNKNOWN")
        critical_count = len(validation.get("critical_failures", []))
        warning_count = len(validation.get("warnings", []))
        info_count = len(validation.get("info_notes", []))
        
        if status == "PASS":
            print(f"✓ Validation PASSED")
            print(f"  - Critical failures: {critical_count}")
            print(f"  - Warnings: {warning_count}")
            print(f"  - Info: {info_count}")
            test_results["modules"]["07_validation"] = {
                "status": "PASS",
                "critical": critical_count,
                "warnings": warning_count,
                "info": info_count
            }
        else:
            print(f"✗ Validation FAILED")
            for error in validation.get("critical_failures", []):
                print(f"    CRITICAL: {error}")
            test_results["modules"]["07_validation"] = {
                "status": "FAIL",
                "critical_failures": validation.get("critical_failures", [])
            }
            return False
    
    except Exception as e:
        print(f"✗ FAILED: {e}")
        test_results["modules"]["07_validation"] = {
            "status": "FAIL",
            "error": str(e)
        }
        return False
    
    # Summary
    print("\n" + "=" * 70)
    print("ALL TESTS PASSED ✓")
    print("=" * 70)
    
    # Save test results
    save_json(f"logs/test_run_{test_date}.json", test_results)
    
    print(f"\nNext steps:")
    print("1. Set up Google Sheets + Drive API")
    print("2. Add Modules 08–12 (Google Sheets write, verification, seeding, extraction, logging)")
    print("3. Test full pipeline with workbook integration")
    print("4. Deploy to Replit with mobile UI")
    
    return True

if __name__ == "__main__":
    success = run_test_pipeline()
    sys.exit(0 if success else 1)
